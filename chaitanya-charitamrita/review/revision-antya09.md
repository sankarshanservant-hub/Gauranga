```python
# conv_en.py <stageA.txt> <first> <last> [ov.py]  → печатает блоки «## N / w: / tr: / fn:» в stdout
import re, sys, runpy
src, a, b = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
ov = runpy.run_path(sys.argv[4]) if len(sys.argv) > 4 else {}
TR, W, FN = ov.get('TR', {}), ov.get('W', {}), ov.get('FN', {})
blocks = {}; cur = None
for l in open(src, encoding='utf-8').read().split('\n'):
    if l.startswith('#') or l.startswith('@title'): continue
    m = re.match(r'@(\d+(?:–\d+)?)\s*$', l)
    if m: cur = m.group(1); blocks[cur] = {'w': [], 'tr': None}; continue
    if cur is None or not l.strip(): continue
    if l.startswith('= '): blocks[cur]['tr'] = l[2:].strip()
    else: blocks[cur]['w'].append(l.strip())
out = []
for k, v in blocks.items():
    n = int(k.split('–')[0])
    if not (a <= n <= b): continue
    out.append(f'## {k}'); out += [f'w: {x}' for x in W.get(n, v['w'])]
    out.append(f"tr: {TR.get(n, v['tr'])}"); out += [f'fn: {x}' for x in FN.get(n, [])]; out.append('')
print('\n'.join(out))
```
